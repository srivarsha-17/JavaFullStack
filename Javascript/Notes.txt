    Array inbuilt functions : 
    let arr1 = [null,true,500,'javascript']

    1.push():
      used to push element at the end of the array 

      Eg: arr1.push(50) => [null,true,500,'js',50] => added at the end


    2.spread operator (...):
    Eg : arr1 = [10,20,30,40]
        arr2 = [1,2,3,4,5]
        arr2.push(...arr1)
        now arr2 = [1,2,3,4,5,10,20,30,40,50] // arr2 is added to arr1

        arr2.push(arr1) // the entire array is added as a new element

        now arr2 = [1,2,3,4,5,[10,20,30,40,50]]

    shift opertor is used to remove element at the 0th position

for-of loop:  used to print the values of the array
--------------------------------------------------------
let arr = [10,20,30,40,50]

let str = "Javascript"


for(let val of arr){
  console.log(val);   //used to print the values of the array
}

for(let val of str){
  console.log(val); //prints each character of the string
}



for-in loop: used to fetch the index value of a array/ string
--------------------------------------------------------

for(let index in arr){
  console.log(index)  //fetches the index values
}


for(let index in arr){
  console.log(arr[index])  //we are printing the values of the arr using the index 
}


 
forEach loop: used to fetch both the arrays and the values  (Introducted in ES6)
-------------------------------------------------------------------------------------------------------------------------
arr.forEach((val,ind,a)=>{  //val stores the value , ind stores the index and arr stands for the array
  console.log(val+" "+ind+" "+a);
})
